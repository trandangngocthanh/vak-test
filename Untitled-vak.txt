  /* ---------- Chấm điểm ---------- */
  function compute() {
    var c = { V: 0, A: 0, K: 0 };
    answers.forEach(function (a) { if (a) c[a]++; });
    var n = answers.filter(Boolean).length || 1;
    // làm tròn sao cho tổng đúng 100%
    var parts = CH.map(function (k) { var v = c[k] / n * 100; return { k: k, f: Math.floor(v), r: v - Math.floor(v) }; });
    var left = 100 - parts.reduce(function (s, p) { return s + p.f; }, 0);
    parts.slice().sort(function (a, b) { return b.r - a.r; }).forEach(function (p) { if (left > 0) { p.f++; left--; } });
    var pct = {}; parts.forEach(function (p) { pct[p.k] = p.f; });

    var sorted = CH.slice().sort(function (a, b) { return c[b] - c[a]; });
    var kind, keys;
    if (c[sorted[0]] - c[sorted[2]] <= 2) { kind = 'balanced'; keys = sorted; }
    else if (c[sorted[0]] - c[sorted[1]] <= 1) { kind = 'combo'; keys = sorted.slice(0, 2); }
    else { kind = 'single'; keys = [sorted[0]]; }
    var label = kind === 'balanced' ? 'Đa kênh (V–A–K cân bằng)'
      : kind === 'combo' ? keys.join('+') + ' – ' + keys.map(function (k) { return LABEL[k]; }).join(' & ')
      : keys[0] + ' – ' + LABEL[keys[0]];
    return { c: c, pct: pct, kind: kind, keys: keys, label: label, sorted: sorted };
  }

  function normPhone(p) {
    var s = String(p).replace(/[\s.\-()]/g, '');
    if (s.indexOf('+84') === 0) s = '0' + s.slice(3);
    else if (s.indexOf('84') === 0 && s.length === 11) s = '0' + s.slice(2);
    return s;
  }

  $('#lead-form').addEventListener('submit', function (e) {
    e.preventDefault();
    if (sending) return;
    var name = $('#f-name').value.trim().replace(/\s+/g, ' ');
    var phone = normPhone($('#f-phone').value);
    var email = $('#f-email').value.trim().toLowerCase();
    var okName = name.length >= 2;
    var okPhone = /^0(3|5|7|8|9)\d{8}$/.test(phone);
    var okEmail = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email);
    var okConsent = $('#f-consent').checked;
    setInvalid('#fld-name', !okName); setInvalid('#fld-phone', !okPhone);
    setInvalid('#fld-email', !okEmail); setInvalid('#fld-consent', !okConsent);
    if (!(okName && okPhone && okEmail && okConsent)) {
      var first = document.querySelector('.field.invalid input') || $('#f-consent');
      first.focus(); return;
    }

    sending = true;
    var btn = $('#btn-submit'); btn.disabled = true; btn.textContent = 'Đang xử lý…';
    var r = compute();
    var params = new URLSearchParams(location.search);
    var payload = {
      name: name, phone: phone, email: email,
      resultLabel: r.label,
      pctV: r.pct.V, pctA: r.pct.A, pctK: r.pct.K,
      scoreV: r.c.V, scoreA: r.c.A, scoreK: r.c.K,
      durationSec: Math.round((Date.now() - startedAt) / 1000),
      utm_source: params.get('utm_source') || '', utm_medium: params.get('utm_medium') || '',
      utm_campaign: params.get('utm_campaign') || '', utm_content: params.get('utm_content') || '',
      referrer: document.referrer || '',
      answers: answers.map(function (a, i) { return (i + 1) + a; }).join(' '),
      userAgent: navigator.userAgent,

    };
    sendData(payload).then(function () {

      renderResult(name, r);
      sending = false; btn.disabled = false; btn.textContent = 'Xem kết quả của tôi';
      show('screen-result');
      animateViz(r);
    });
  });

  function sendData(payload) {
    if (!CONFIG.scriptUrl) { console.warn('[VAK] Chưa cấu hình scriptUrl – dữ liệu không được lưu.'); return Promise.resolve(); }
    var req = fetch(CONFIG.scriptUrl, {
      method: 'POST', mode: 'no-cors', keepalive: true,
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify(payload)
    }).catch(function (err) { console.error('[VAK] Gửi dữ liệu lỗi:', err); });
    var timeout = new Promise(function (res) { setTimeout(res, 4000); });
    return Promise.race([req, timeout]);
  }

  function drawRing(r, e) {
    var acc = 0, stops = [];
    r.sorted.forEach(function (k) { var b = acc + r.pct[k] * e; stops.push(COLOR[k] + ' ' + acc + '% ' + b + '%'); acc = b; });
    stops.push('rgba(255,255,255,.08) ' + acc + '% 100%');
    $('#r-ring').style.background = 'conic-gradient(' + stops.join(',') + ')';
  }
  function animateViz(r) {
    requestAnimationFrame(function () { requestAnimationFrame(function () {
      document.querySelectorAll('#r-bars .fill').forEach(function (f) { f.style.width = f.getAttribute('data-w') + '%'; });
    }); });
    if (reduceMotion) { drawRing(r, 1); return; }
    var t0 = null, dur = 1200;
    function frame(ts) {
      if (!t0) t0 = ts;
      var p = Math.min(1, (ts - t0) / dur), e = 1 - Math.pow(1 - p, 3);
      drawRing(r, e);
      if (p < 1) requestAnimationFrame(frame);
    }
    requestAnimationFrame(frame);
  }
